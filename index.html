// Live search dropdown supporting major cities, states, and countries searchInput.addEventListener('input', (e) => { clearTimeout(searchTimeout); const query = e.target.value.trim(); if (query.length < 2) { dropdown.style.display = 'none'; return; } searchTimeout = setTimeout(async () => { try { // Fetch top 10 results sorted or filtered naturally const res = await fetch(`https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(query)}&count=10`); const data = await res.json(); dropdown.innerHTML = ''; if (!data.results || data.results.length === 0) { dropdown.style.display = 'none'; return; } // Sort by population so major cities and primary states appear at the top const sortedResults = data.results.sort((a, b) => (b.population || 0) - (a.population || 0)); sortedResults.forEach(loc => { const item = document.createElement('div'); item.className = 'dropdown-item'; // Format display as: "City/State, Region, Country" const region = loc.admin1 ? `, ${loc.admin1}` : ''; const country = loc.country ? `, ${loc.country}` : ''; const displayName = `${loc.name}${region}${country}`; item.textContent = displayName; item.addEventListener('click', () => { searchInput.value = displayName; dropdown.style.display = 'none'; fetchWeatherData(loc.latitude, loc.longitude, displayName, loc.timezone); }); dropdown.appendChild(item); }); dropdown.style.display = 'block'; } catch (err) { console.error(err); } }, 300); });
