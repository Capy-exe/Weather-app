// Live search dropdown limited strictly to States, Union Territories, and Countries searchInput.addEventListener('input', (e) => { clearTimeout(searchTimeout); const query = e.target.value.trim(); if (query.length < 2) { dropdown.style.display = 'none'; return; } searchTimeout = setTimeout(async () => { try { // Fetch top 15 results to scan for administrative areas const res = await fetch(`https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(query)}&count=15`); const data = await res.json(); dropdown.innerHTML = ''; if (!data.results || data.results.length === 0) { dropdown.style.display = 'none'; return; } // Filter for primary administrative divisions (States, UTs, Regions, Countries) // Feature codes like ADM1 (State/Province), ADM2, PCL (Country), etc. const filteredResults = data.results.filter(loc => { const code = loc.feature_code || ''; // Keep major administrative regions (ADM1, ADM2) or country level (PCLI, PCL) return code.startsWith('ADM') || code.startsWith('PCL') || loc.name.toLowerCase() === query.toLowerCase(); }); // If no strict admin match found, fallback to top population results const displayList = filteredResults.length > 0 ? filteredResults : data.results.slice(0, 5); displayList.forEach(loc => { const item = document.createElement('div'); item.className = 'dropdown-item'; // Format display as: "State/Region, Country" const country = loc.country ? `, ${loc.country}` : ''; const displayName = `${loc.name}${country}`; item.textContent = displayName; item.addEventListener('click', () => { searchInput.value = displayName; dropdown.style.display = 'none'; fetchWeatherData(loc.latitude, loc.longitude, displayName, loc.timezone); }); dropdown.appendChild(item); }); dropdown.style.display = 'block'; } catch (err) { console.error(err); } }, 300); });
